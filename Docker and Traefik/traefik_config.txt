Pipe SMS Container
============================================

docker run -it -d \
  --label traefik.http.routers.pipesms.rule=Host\(\`pipesms.aifrruislabs.com\`\) \
  --label traefik.http.routers.pipesms.tls.certresolver=lets-encrypt \
  --label traefik.http.routers.pipesms.tls=true \
  --label traefik.http.routers.pipesms.service=pipesms-http-service \
  --label traefik.http.services.pipesms-http-service.loadbalancer.server.port=80 \
  --name pipesms_container \
  --network web \
  --expose 80 \
   emasanga/pipesms:latest
